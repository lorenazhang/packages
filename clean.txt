SELECT
    DatabaseName,
    TableName,
    MAX(CASE 
        WHEN LOWER(ColumnName) LIKE '%recontact%' 
        THEN ColumnName 
    END) AS recontact_col,
    MAX(CASE 
        WHEN LOWER(ColumnName) LIKE '%conn%' 
        THEN ColumnName 
    END) AS conn_col
FROM DBC.ColumnsV
WHERE DatabaseName = 'sbx_ceda_poc'
  AND LOWER(TableName) LIKE 'ccr%'
  AND (
      LOWER(ColumnName) LIKE '%recontact%'
      OR LOWER(ColumnName) LIKE '%conn%'
  )
GROUP BY
    DatabaseName,
    TableName
HAVING
    MAX(CASE WHEN LOWER(ColumnName) LIKE '%recontact%' THEN 1 ELSE 0 END) = 1
    AND
    MAX(CASE WHEN LOWER(ColumnName) LIKE '%conn%' THEN 1 ELSE 0 END) = 1
ORDER BY
    TableName;
