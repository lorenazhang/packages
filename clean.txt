import pandas as pd
from datetime import date, timedelta

# ============================================================
# 1. Date range: last 24 weeks
# ============================================================

today = date.today()

this_monday = today - timedelta(days=today.weekday())
this_monday_str = this_monday.strftime("%Y-%m-%d")

start_date = this_monday - timedelta(weeks=24)
start_date_str = start_date.strftime("%Y-%m-%d")

print("start_date:", start_date_str)
print("end_date:", this_monday_str)


# ============================================================
# 2. Recontact baseline query - TDAP2
# ============================================================

query = f"""
WITH call_data AS
(
    SELECT
        LOWER(TRIM(a.connid)) AS connid,
        a.interactionid,
        a.calldriver1,
        a.aa_virtualqueue_bb_cct AS virtual_queue,
        a.interactiondatetime,

        /* Monday week start */
        CAST(SUBSTR(a.interactiondatetime, 1, 10)
             AS DATE FORMAT 'YYYY-MM-DD')
        -
        (
            (
                CAST(SUBSTR(a.interactiondatetime, 1, 10)
                     AS DATE FORMAT 'YYYY-MM-DD')
                - DATE '1900-01-01'
            ) MOD 7
        ) AS week_start_call

    FROM sbx_ceda_poc.call_driver_output_v2 a

    WHERE a.model_lang = 'English'
      AND a.calldriver1 NOT IN (
          'No Conversation',
          'No Reason Disclosed'
      )
      AND a.connid IS NOT NULL
      AND a.aa_virtualqueue_bb_cct IS NOT NULL

      AND a.interactiondatetime >= '{start_date_str}'
      AND a.interactiondatetime < '{this_monday_str}'

      AND SUBSTR(a.interactiondatetime, 1, 10)
          LIKE '____-__-__'

    QUALIFY ROW_NUMBER() OVER
    (
        PARTITION BY LOWER(TRIM(a.connid))
        ORDER BY a.interactiondatetime
    ) = 1
),

call_recontact AS
(
    SELECT
        c.*,

        CASE
            WHEN EXISTS
            (
                SELECT 1
                FROM sbx_ceda_poc.ccr_fact_voc_recontacts_v3_prd r
                WHERE LOWER(TRIM(r.voc_id_parent)) = c.connid
                  AND r.voc_id_parent IS NOT NULL
            )
            THEN 1
            ELSE 0
        END AS recontact_ind

    FROM call_data c
)

SELECT
    week_start_call,
    calldriver1,
    virtual_queue,

    COUNT(*) AS call_count,

    SUM(recontact_ind) AS recontact_count,

    CAST(SUM(recontact_ind) AS DECIMAL(18,6))
        / NULLIFZERO(COUNT(*))
        AS recontact_rate

FROM call_recontact

GROUP BY
    week_start_call,
    calldriver1,
    virtual_queue

ORDER BY
    week_start_call,
    calldriver1,
    virtual_queue
"""
