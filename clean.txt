df['createdtime'] = pd.to_datetime(df['createdtime'])

df['week'] = df['createdtime'].dt.to_period('W').apply(lambda x: x.start_time)
df['month'] = df['createdtime'].dt.strftime('%Y%m')

weekly = (
    df.groupby(['week', 'social_network'])
      .size()
      .reset_index(name='post_count')
)

import matplotlib.pyplot as plt

plt.figure(figsize=(14, 6))

for platform, temp in weekly.groupby('social_network'):
    plt.plot(
        temp['week'],
        temp['post_count'],
        marker='o',
        label=platform
    )

plt.xlabel('Week')
plt.ylabel('Post Count')
plt.title('Weekly Social Media Post Volume')
plt.legend()
plt.xticks(rotation=45)
plt.tight_layout()
plt.show()
