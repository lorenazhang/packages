import pandas as pd
import numpy as np

# ============================================================
# Target table
# ============================================================

target_table = "cdar_cai_prd.eo_recontact_13m_baseline"


# ============================================================
# Prepare dataframe
# ============================================================

df_load = df_final.copy()

df_load['week_start_call'] = pd.to_datetime(
    df_load['week_start_call']
).dt.date

df_load['call_count'] = df_load['call_count'].astype(int)
df_load['recontact_count'] = df_load['recontact_count'].astype(int)
df_load['recontact_rate'] = df_load['recontact_rate'].astype(float)

# Replace NaN with None so Teradata inserts them as NULL
df_load = df_load.replace({np.nan: None})


# ============================================================
# Create target table
# ============================================================

cursor = conn_tdap2.cursor()

create_sql = f"""
CREATE MULTISET TABLE {target_table}
(
    week_start_call DATE,
    calldriver1 VARCHAR(200),
    primary_lob VARCHAR(200),
    sublob VARCHAR(300),
    call_count INTEGER,
    recontact_count INTEGER,
    recontact_rate DECIMAL(18,6)
)
PRIMARY INDEX (
    week_start_call,
    calldriver1
)
"""

cursor.execute(create_sql)

print(f"Created table: {target_table}")


# ============================================================
# Insert data
# ============================================================

insert_sql = f"""
INSERT INTO {target_table}
(
    week_start_call,
    calldriver1,
    primary_lob,
    sublob,
    call_count,
    recontact_count,
    recontact_rate
)
VALUES (?, ?, ?, ?, ?, ?, ?)
"""

rows = list(
    df_load[
        [
            'week_start_call',
            'calldriver1',
            'primary_lob',
            'sublob',
            'call_count',
            'recontact_count',
            'recontact_rate'
        ]
    ].itertuples(index=False, name=None)
)

batch_size = 5000

for i in range(0, len(rows), batch_size):

    batch = rows[i:i + batch_size]

    cursor.executemany(
        insert_sql,
        batch
    )

    print(
        f"Loaded {min(i + batch_size, len(rows)):,} "
        f"/ {len(rows):,}"
    )


# ============================================================
# Validate loaded table
# ============================================================

check_query = f"""
SELECT
    COUNT(*) AS row_count,
    MIN(week_start_call) AS min_week,
    MAX(week_start_call) AS max_week
FROM {target_table}
"""

df_check = pd.read_sql(
    check_query,
    conn_tdap2
)

print(df_check)
