WITH call_raw AS
(
    SELECT
        LOWER(TRIM(connid2)) AS connid2,
        LOWER(TRIM(globalid)) AS globalid,
        interactionid,
        aa_virtualqueue_bb_cct,
        calldriver1,
        interactiondatetime,

        CAST(
            SUBSTR(interactiondatetime, 1, 10)
            AS DATE FORMAT 'YYYY-MM-DD'
        )
        -
        (
            (
                CAST(
                    SUBSTR(interactiondatetime, 1, 10)
                    AS DATE FORMAT 'YYYY-MM-DD'
                )
                - DATE '1900-01-01'
            ) MOD 7
        ) AS week_start_call

    FROM EIW_U_CEDA_VOCAL_V1.AMP_CALLDRIVERS2_0

    WHERE model_lang = 'English'
      AND calldriver1 NOT IN (
          'No Conversation',
          'No Reason Disclosed'
      )
      AND (connid2 IS NOT NULL OR globalid IS NOT NULL)
      AND aa_virtualqueue_bb_cct IS NOT NULL
      AND interactiondatetime >= '2025-05-26'
      AND interactiondatetime <  '2026-07-27'
      AND SUBSTR(interactiondatetime, 1, 10) LIKE '____-__-__'
),


/* -------------------------
   Old / migrated calls
   ------------------------- */

call_connid AS
(
    SELECT *
    FROM call_raw
    WHERE connid2 IS NOT NULL

    QUALIFY ROW_NUMBER() OVER (
        PARTITION BY connid2
        ORDER BY interactiondatetime
    ) = 1
),


/* -------------------------
   New-system calls
   ------------------------- */

call_globalid AS
(
    SELECT *
    FROM call_raw
    WHERE connid2 IS NULL
      AND globalid IS NOT NULL

    QUALIFY ROW_NUMBER() OVER (
        PARTITION BY globalid
        ORDER BY interactiondatetime
    ) = 1
),


comp_data AS
(
    SELECT
        LOWER(TRIM(conn_id)) AS conn_id,
        LOWER(TRIM(global_id)) AS global_id,
        case_nr,
        complaint_type_desc

    FROM EIW_I_CEDA_S1.COMPLAINTS_ECMP_REGISTRY_DLY_CS

    WHERE wf_recd_dt >= DATE '2025-05-26'
      AND wf_recd_dt <  DATE '2026-08-03'
),


/* -------------------------
   Join by CONNID
   ------------------------- */

joined_connid AS
(
    SELECT
        c.week_start_call,
        c.calldriver1,
        c.aa_virtualqueue_bb_cct,
        c.connid2 AS call_id,
        p.case_nr,
        p.complaint_type_desc

    FROM call_connid c

    LEFT JOIN comp_data p
        ON c.connid2 = p.conn_id
),


/* -------------------------
   Join by GLOBAL ID
   ------------------------- */

joined_globalid AS
(
    SELECT
        c.week_start_call,
        c.calldriver1,
        c.aa_virtualqueue_bb_cct,
        c.globalid AS call_id,
        p.case_nr,
        p.complaint_type_desc

    FROM call_globalid c

    LEFT JOIN comp_data p
        ON c.globalid = p.global_id
),


/* -------------------------
   Combine
   ------------------------- */

joined_data AS
(
    SELECT * FROM joined_connid

    UNION ALL

    SELECT * FROM joined_globalid
)


SELECT
    week_start_call,
    calldriver1,
    aa_virtualqueue_bb_cct,

    COUNT(*) AS call_count,

    SUM(
        CASE WHEN case_nr IS NOT NULL
        THEN 1 ELSE 0 END
    ) AS complaint_count,

    SUM(
        CASE
            WHEN complaint_type_desc = 'Service Complaint'
            THEN 1 ELSE 0
        END
    ) AS service_complaint_count,

    SUM(
        CASE
            WHEN complaint_type_desc IN (
                'LOB Escalated Complaint',
                'EO Escalated Complaint'
            )
            THEN 1 ELSE 0
        END
    ) AS escalated_complaint_count,

    CAST(
        SUM(CASE WHEN case_nr IS NOT NULL THEN 1 ELSE 0 END)
        AS DECIMAL(18,6)
    ) / NULLIFZERO(COUNT(*))
        AS complaint_conversion_rate,

    CAST(
        SUM(
            CASE
                WHEN complaint_type_desc = 'Service Complaint'
                THEN 1 ELSE 0
            END
        )
        AS DECIMAL(18,6)
    ) / NULLIFZERO(COUNT(*))
        AS service_conversion_rate,

    CAST(
        SUM(
            CASE
                WHEN complaint_type_desc IN (
                    'LOB Escalated Complaint',
                    'EO Escalated Complaint'
                )
                THEN 1 ELSE 0
            END
        )
        AS DECIMAL(18,6)
    ) / NULLIFZERO(COUNT(*))
        AS escalated_conversion_rate

FROM joined_data

GROUP BY
    week_start_call,
    calldriver1,
    aa_virtualqueue_bb_cct
;
