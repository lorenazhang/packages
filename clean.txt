import numpy as np
import pandas as pd

group_cols = ['calldriver1', 'primary_lob', 'sublob']
date_col = 'week_start_call'
rate_col = 'conversion_rate'

df_sorted = df.sort_values(group_cols + [date_col])


# 1. 每个 calldriver1 × LOB × SubLOB 提取最近 6 周的 rate 向量
recent_6wk = (
    df_sorted
    .groupby(group_cols, dropna=False)
    .tail(6)
    .groupby(group_cols, dropna=False)
    .agg(
        recent_6wk_dates=(date_col, list),
        recent_6wk_rates=(rate_col, list),
        n_weeks=(rate_col, 'count')
    )
    .reset_index()
)


# 2. 根据最近 6 周 rate 计算 slope
def calculate_slope(rates):
    rates = np.asarray(rates, dtype=float)

    # 必须有完整 6 周，并且不能有缺失值
    if len(rates) != 6 or np.isnan(rates).any():
        return np.nan

    x = np.arange(6)

    return np.polyfit(x, rates, 1)[0]


recent_6wk['recent_6wk_slope'] = (
    recent_6wk['recent_6wk_rates']
    .apply(calculate_slope)
)

# slope 转成每周变化的百分点，方便解释
recent_6wk['recent_6wk_slope_pct_point'] = (
    recent_6wk['recent_6wk_slope'] * 100
)
