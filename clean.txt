SELECT
    b.*,
    
    md.conn_id AS metadata_conn_id,
    md.interaction_id,
    
    CASE 
        WHEN md.conn_id IS NOT NULL THEN 1 
        ELSE 0 
    END AS conn_match_ind,

    CASE 
        WHEN tr.interaction_id IS NOT NULL THEN 1 
        ELSE 0 
    END AS interaction_match_ind,

    tr.*

FROM call_driver_cases b

LEFT JOIN SBX_CEDA_POC.wf_lob_byodb_metadata md
    ON LOWER(TRIM(b.conn_id)) = LOWER(TRIM(md.conn_id))

LEFT JOIN SBX_CEDA_POC.wf_lob_byodb_full tr
    ON LOWER(TRIM(md.interaction_id)) = LOWER(TRIM(tr.interaction_id));
