recontact_data AS
(
    SELECT DISTINCT
        LOWER(TRIM(voc_id_parent)) AS connid

    FROM sbx_ceda_poc.ccr_fact_voc_recontacts_v3_prd

    WHERE voc_id_parent IS NOT NULL
      AND voc_datetime >= DATE '{start_date_str}'
),

call_recontact AS
(
    SELECT
        c.*,

        CASE
            WHEN r.connid IS NOT NULL THEN 1
            ELSE 0
        END AS recontact_ind

    FROM call_data c

    LEFT JOIN recontact_data r
        ON c.connid = r.connid
)
