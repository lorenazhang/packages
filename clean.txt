# Ensure date column is datetime

df['week_start_call'] = pd.to_datetime(df['week_start_call'])

# Create month column

df['month'] = df['week_start_call'].dt.to_period('M')

# Calculate null rate by month

null_rate = (

    df.groupby('month')['division_name']

      .apply(lambda x: x.isna().mean())

      .reset_index(name='null_rate')

)

# Convert month back to string if desired

null_rate['month'] = null_rate['month'].astype(str)

print(null_rate)
