
group_cols = [
    'month',
    'calldriver1',
    'primary_lob',
    'sublob'
]

count_cols = [
    'call_count',
    'complaint_count',
    'service_complaint_count',
    'escalated_complaint_count'
]

df_monthly = (
    df.groupby(group_cols, dropna=False)[count_cols]
      .sum()
      .reset_index()
)

# recalculate monthly conversion rates
df_monthly['complaint_conversion_rate'] = (
    df_monthly['complaint_count'] / df_monthly['call_count']
)

df_monthly['service_conversion_rate'] = (
    df_monthly['service_complaint_count'] / df_monthly['call_count']
)

df_monthly['escalated_conversion_rate'] = (
    df_monthly['escalated_complaint_count'] / df_monthly['call_count']
)
