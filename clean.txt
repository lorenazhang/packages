WITH call_raw AS
(
    SELECT
        /* standardized IDs */
        LOWER(TRIM(connid2)) AS connid2,
        LOWER(TRIM(globalid)) AS globalid,

        interactionid,
        aa_virtualqueue_bb_cct,
        calldriver1,
        interactiondatetime,

        /* YYYYMM */
        SUBSTR(interactiondatetime, 1, 4)
        || SUBSTR(interactiondatetime, 6, 2) AS month_id,

        /* YYYYMMDD */
        SUBSTR(interactiondatetime, 1, 4)
        || SUBSTR(interactiondatetime, 6, 2)
        || SUBSTR(interactiondatetime, 9, 2) AS day_id,

        /* Monday week start */
        CAST(
            SUBSTR(interactiondatetime, 1, 10)
            AS DATE FORMAT 'YYYY-MM-DD'
        )
        -
        (
            (
                CAST(
                    SUBSTR(interactiondatetime, 1, 10)
                    AS DATE FORMAT 'YYYY-MM-DD'
                )
                - DATE '1900-01-01'
            ) MOD 7
        ) AS week_start_call

    FROM EIW_U_CEDA_VOCAL_V1.AMP_CALLDRIVERS2_0

    WHERE model_lang = 'English'
      AND calldriver1 NOT IN (
            'No Conversation',
            'No Reason Disclosed'
      )

      /* Need at least one usable ID */
      AND (
            connid2 IS NOT NULL
            OR globalid IS NOT NULL
      )

      AND aa_virtualqueue_bb_cct IS NOT NULL

      AND interactiondatetime >= '2025-05-26'
      AND interactiondatetime <  '2026-07-27'

      /* Exclude malformed dates */
      AND SUBSTR(interactiondatetime, 1, 10)
          LIKE '____-__-__'
),


/* ============================================================
   One call may have multiple interactions.
   Deduplicate so each call is counted once.
   Prefer connid2 when available; otherwise use globalid.
   ============================================================ */

call_data AS
(
    SELECT *

    FROM call_raw

    QUALIFY ROW_NUMBER() OVER
    (
        PARTITION BY
            CASE
                WHEN connid2 IS NOT NULL
                    THEN 'C_' || connid2
                ELSE 'G_' || globalid
            END

        ORDER BY interactiondatetime
    ) = 1
),


comp_data AS
(
    SELECT
        LOWER(TRIM(conn_id)) AS conn_id,
        LOWER(TRIM(global_id)) AS global_id,

        case_nr,
        complaint_id,
        complaint_type_desc,

        CAST(wf_recd_dt AS DATE) AS wf_recd_dt,
        CAST(created_dt AS DATE) AS created_dt

    FROM EIW_I_CEDA_S1.COMPLAINTS_ECMP_REGISTRY_DLY_CS

    WHERE (
            conn_id IS NOT NULL
            OR global_id IS NOT NULL
          )

      AND wf_recd_dt >= DATE '2025-05-26'
      AND wf_recd_dt <  DATE '2026-08-03'
)


SELECT
    c.week_start_call,
    c.calldriver1,
    c.aa_virtualqueue_bb_cct,

    /* Counts */
    COUNT(*) AS call_count,

    SUM(
        CASE
            WHEN p.case_nr IS NOT NULL
            THEN 1 ELSE 0
        END
    ) AS complaint_count,

    SUM(
        CASE
            WHEN p.complaint_type_desc = 'Service Complaint'
            THEN 1 ELSE 0
        END
    ) AS service_complaint_count,

    SUM(
        CASE
            WHEN p.complaint_type_desc IN (
                'LOB Escalated Complaint',
                'EO Escalated Complaint'
            )
            THEN 1 ELSE 0
        END
    ) AS escalated_complaint_count,


    /* Conversion rates */
    CAST(
        SUM(
            CASE
                WHEN p.case_nr IS NOT NULL
                THEN 1 ELSE 0
            END
        ) AS DECIMAL(18,6)
    )
    / NULLIFZERO(COUNT(*))
        AS complaint_conversion_rate,


    CAST(
        SUM(
            CASE
                WHEN p.complaint_type_desc = 'Service Complaint'
                THEN 1 ELSE 0
            END
        ) AS DECIMAL(18,6)
    )
    / NULLIFZERO(COUNT(*))
        AS service_conversion_rate,


    CAST(
        SUM(
            CASE
                WHEN p.complaint_type_desc IN (
                    'LOB Escalated Complaint',
                    'EO Escalated Complaint'
                )
                THEN 1 ELSE 0
            END
        ) AS DECIMAL(18,6)
    )
    / NULLIFZERO(COUNT(*))
        AS escalated_conversion_rate,


    /* Average complaint gap */
    AVG(
        CASE
            WHEN p.created_dt IS NOT NULL
             AND p.wf_recd_dt IS NOT NULL
            THEN p.created_dt - p.wf_recd_dt
            ELSE NULL
        END
    ) AS avg_gap_days


FROM call_data c

LEFT JOIN comp_data p
    ON
    (
        /* connid2 takes priority */
        c.connid2 IS NOT NULL
        AND c.connid2 = p.conn_id
    )

    OR

    (
        /* Only use globalid when connid2 is unavailable */
        c.connid2 IS NULL
        AND c.globalid IS NOT NULL
        AND c.globalid = p.global_id
    )


GROUP BY
    c.week_start_call,
    c.calldriver1,
    c.aa_virtualqueue_bb_cct


ORDER BY
    c.calldriver1,
    c.aa_virtualqueue_bb_cct,
    c.week_start_call
;
