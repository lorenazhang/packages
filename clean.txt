import numpy as np

num_of_month = 4

group_cols = [
    'calldriver1',
    'primary_lob',
    'sublob'
]

# make sure sorted by month
df_monthly = df_monthly.sort_values(group_cols + ['month'])


def calculate_recent_slope(group, num_of_month=4):
    
    # take most recent N months
    recent = group.tail(num_of_month)
    
    rates = recent['complaint_conversion_rate'].values
    
    # require at least 3 months
    if len(rates) < 3 or np.isnan(rates).any():
        return np.nan
    
    x = np.arange(len(rates))
    
    return np.polyfit(x, rates, 1)[0]


slope_df = (
    df_monthly
    .groupby(group_cols, dropna=False)
    .apply(
        lambda x: calculate_recent_slope(
            x,
            num_of_month=num_of_month
        )
    )
    .reset_index(name='slope')
)
