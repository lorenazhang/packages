num_of_months = 3

def calculate_trend(group, num_of_months=3):
    
    group = group.sort_values('month')
    
    # current month + previous N months
    recent = group.tail(num_of_months + 1)
    
    rates = recent['complaint_conversion_rate'].to_numpy(dtype=float)

    if len(rates) < num_of_months + 1 or np.isnan(rates).any():
        return pd.Series({
            'continuous_uptrend': False,
            'rate_change': np.nan
        })

    # every month must increase
    continuous_uptrend = np.all(np.diff(rates) > 0)

    # latest month - starting month
    rate_change = rates[-1] - rates[0]

    return pd.Series({
        'continuous_uptrend': continuous_uptrend,
        'rate_change': rate_change
    })


trend_df = (
    df_monthly
    .groupby(
        ['calldriver1', 'primary_lob', 'sublob'],
        dropna=False
    )
    .apply(
        calculate_trend,
        num_of_months=num_of_months
    )
    .reset_index()
)
