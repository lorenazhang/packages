
num_of_month = 4

def calculate_trend(group, num_of_month=4):
    
    recent = (
        group.sort_values('month')
             .tail(num_of_month)
    )

    rates = recent['complaint_conversion_rate'].to_numpy(dtype=float)

    if len(rates) < 3 or np.isnan(rates).any():
        return pd.Series({
            'slope': np.nan,
            'positive_change_pct': np.nan
        })

    # linear slope
    x = np.arange(len(rates))
    slope = np.polyfit(x, rates, 1)[0]

    # month-over-month changes
    changes = np.diff(rates)

    # % of changes that are positive
    positive_change_pct = (changes > 0).mean()

    return pd.Series({
        'slope': slope,
        'positive_change_pct': positive_change_pct
    })


trend_df = (
    df_monthly
    .groupby(
        ['calldriver1', 'primary_lob', 'sublob'],
        dropna=False
    )
    .apply(
        calculate_trend,
        num_of_month=num_of_month
    )
    .reset_index()
)
