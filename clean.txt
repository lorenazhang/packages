#create volatile table in eiwp
cursor = conn_eiwp.cursor()

cursor.execute("""
CREATE VOLATILE TABLE vt_recontact
(
    connid VARCHAR(100)
)
PRIMARY INDEX (connid)
ON COMMIT PRESERVE ROWS
""")

insert_sql = """
INSERT INTO vt_recontact (connid)
VALUES (?)
"""

rows = [(x,) for x in df_recontact['connid'].dropna().unique()]

batch_size = 10000

for i in range(0, len(rows), batch_size):
    cursor.executemany(
        insert_sql,
        rows[i:i + batch_size]
    )

print(f"Inserted {len(rows):,} unique recontact conn_ids")

query = f"""
WITH call_data AS
(
    SELECT
        LOWER(TRIM(a.connid)) AS connid,
        a.interactionid,
        a.calldriver1,
        a.aa_virtualqueue_bb_cct AS virtual_queue,
        a.interactiondatetime,

        /* YYYYMM */
        SUBSTR(a.interactiondatetime, 1, 4)
        || SUBSTR(a.interactiondatetime, 6, 2) AS month_id,

        /* Monday week start */
        CAST(SUBSTR(a.interactiondatetime, 1, 10)
             AS DATE FORMAT 'YYYY-MM-DD')
        -
        (
            (
                CAST(SUBSTR(a.interactiondatetime, 1, 10)
                     AS DATE FORMAT 'YYYY-MM-DD')
                - DATE '1900-01-01'
            ) MOD 7
        ) AS week_start_call

    FROM {call_table} a

    WHERE a.model_lang = 'English'
      AND a.calldriver1 NOT IN ('No Conversation', 'No Reason Disclosed')
      AND a.connid IS NOT NULL
      AND a.aa_virtualqueue_bb_cct IS NOT NULL

      AND a.interactiondatetime >= '{start_date}'
      AND a.interactiondatetime <  '{end_date}'

      /* exclude malformed datetime */
      AND SUBSTR(a.interactiondatetime, 1, 10)
          LIKE '____-__-__'

    QUALIFY ROW_NUMBER() OVER
    (
        PARTITION BY LOWER(TRIM(a.connid))
        ORDER BY a.interactiondatetime
    ) = 1
),

call_recontact AS
(
    SELECT
        c.*,

        CASE
            WHEN r.connid IS NOT NULL THEN 1
            ELSE 0
        END AS recontact_ind

    FROM call_data c

    LEFT JOIN vt_recontact r
        ON c.connid = r.connid
),

vq_mapping AS
(
    SELECT DISTINCT
        LOWER(TRIM(virtual_queue)) AS virtual_queue,
        lob,
        sublob
    FROM {vq_mapping_table}
)

SELECT
    c.week_start_call,
    c.month_id,
    c.calldriver1,
    m.lob,
    m.sublob,

    COUNT(*) AS call_count,

    SUM(c.recontact_ind) AS recontact_count,

    CAST(SUM(c.recontact_ind) AS DECIMAL(18,6))
        / NULLIFZERO(COUNT(*))
        AS recontact_rate

FROM call_recontact c

LEFT JOIN vq_mapping m
    ON LOWER(TRIM(c.virtual_queue)) = m.virtual_queue

GROUP BY
    c.week_start_call,
    c.month_id,
    c.calldriver1,
    m.lob,
    m.sublob

ORDER BY
    c.week_start_call,
    m.lob,
    m.sublob,
    c.calldriver1
"""

df_recontact_baseline = pd.read_sql(query, conn_eiwp)
